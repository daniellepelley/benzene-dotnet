using System.Diagnostics.CodeAnalysis;
using Benzene.Abstractions.DI;
using Benzene.Abstractions.Messages;

namespace Benzene.Service.Clients.UserGet
{
    [ExcludeFromCodeCoverage]
    public static class UserGetServiceClientRegistration
    {
        /// <summary>Registers UserGetServiceClient as IUserGetServiceClient.</summary>
        public static IBenzeneServiceContainer AddUserGetServiceClient(this IBenzeneServiceContainer container)
        {
            // Scoped, not singleton: AddOutboundRouting registers IBenzeneMessageSender
            // scoped, so a singleton client would be a captive dependency.
            container.AddScoped<IUserGetServiceClient, UserGetServiceClient>();
            // The topics this client calls are this service's send declarations
            // (contract-document.md §3.2): they appear in its own contract document's sends.
            return container.AddSingleton<IMessageDefinitionFinder<IMessageSenderDefinition>>(_ => new UserGetServiceClientSends());
        }
    }

    [ExcludeFromCodeCoverage]
    public sealed class UserGetServiceClientSends : IMessageDefinitionFinder<IMessageSenderDefinition>
    {
        public IMessageSenderDefinition[] FindDefinitions() => new IMessageSenderDefinition[]
        {
            new SendDeclaration("user:get", "", typeof(GetUserMessage), typeof(UserDto)),
        };

        private sealed class SendDeclaration : IMessageSenderDefinition, ITopic
        {
            public SendDeclaration(string id, string version, System.Type requestType, System.Type responseType)
            {
                Id = id;
                Version = version;
                RequestType = requestType;
                ResponseType = responseType;
            }

            public string Id { get; }
            public string Version { get; }
            public ITopic Topic => this;
            public System.Type RequestType { get; }
            public System.Type ResponseType { get; }
            public System.Type SenderType => typeof(Benzene.Abstractions.Results.Void);
        }
    }
}
